pipeline {
    agent any
    
    environment {
        GH_CREDS = credentials('github-registry-credentials')
        REGISTRY = 'ghcr.io'
        IMAGE_NAMESPACE = 'ahmedhassan673/visiospark' 
    }

    stages {
        stage('Checkout Code') {
            steps {
                git branch: 'main', url: 'https://github.com/AhmedHassan673/VISIOSPARK.git'
            }
        }

        stage('Build & Push Docker Images to GHCR') {
            steps {
                script {
                    // Log in to GitHub Container Registry
                    sh "echo ${GH_CREDS_PSW} | docker login ghcr.io -u ${GH_CREDS_USR} --password-stdin"
                    
                    // Build & Push Backend
                    appBackend = docker.build("${REGISTRY}/${IMAGE_NAMESPACE}-backend:latest", "./backend")
                    appBackend.push()

                    // Build & Push Frontend
                    appFrontend = docker.build("${REGISTRY}/${IMAGE_NAMESPACE}-frontend:latest", "./frontend")
                    appFrontend.push()
                }
            }
        }

        stage('Deploy to Kubernetes') {
            steps {
                script {
                    // Apply Kubernetes manifests sequentially from your k8s folder
                    sh 'kubectl apply -f k8s/namespace.yaml'
                    sh 'kubectl apply -f k8s/postgres.yaml'
                    sh 'kubectl apply -f k8s/backend.yaml'
                    sh 'kubectl apply -f k8s/frontend.yaml'
                    sh 'kubectl apply -f k8s/ingress.yaml'
                }
            }
        }
    }
    
    post {
        success {
            echo 'Pipeline executed successfully, images pushed to GHCR and deployed to Minikube!'
        }
        failure {
            echo 'Pipeline failed. Check stage logs for details.'
        }
    }
}
